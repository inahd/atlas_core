```python
# Current (correct but incomplete):

# Full encoding (adds Kala position):
def full_phi(element, guna, kala_position=None, domain=None):
    """
    Compute complete φ coordinate.
    
    element:        primary element (earth/water/fire/air/ether)
    guna:           sattva/rajas/tamas
    kala_position:  0.0 (witness) to 1.0 (participant), optional
    domain:         'music'/'ecology'/'geometry' etc, optional
    """
    # Base from element (already encodes left/right)
    # ether → left, earth → right
    element_map = {
        'ether': 0.0,   # far left
        'air':   0.25,  # left
        'water': 0.5,   # center-left
        'fire':  0.75,  # center-right  
        'earth': 1.0,   # right
    }
    base = element_map.get(element.lower(), 0.5)
    
    # Guna modifies within element zone
    guna_offset = {
        'sattva': -0.05,  # pulls left (witness)
        'rajas':   0.05,  # pulls right (action)
        'tamas':   0.0,   # stays
    }
    base += guna_offset.get(guna.lower(), 0)
    
    # Explicit Kalā position overrides if provided
    if kala_position is not None:
        base = kala_position
    
    # Convert to radians
    return (base % 1.0) * 2 * math.pi
```

### The φ-aware coherence function

```python
def coherence_phi_weighted(
    moment_theta, moment_phi,
    entity_theta, entity_phi,
    phi_center=math.pi,
    phi_weight=0.5,
    R=3.0, r=1.0
):
    """
    Coherence weighted by both axes.
    
    phi_weight: 0.0 = only time matters
                0.5 = equal weight (default)
                1.0 = only mode matters
    """
    # Standard toroidal distance
    d_theta = math.pi - abs(abs(moment_theta - entity_theta) - math.pi)
    d_phi   = math.pi - abs(abs(moment_phi   - entity_phi)   - math.pi)
    
    # User's φ position modifies the phi distance
    # If user is at center (π), entities near φ=π score higher
    d_phi_to_user = math.pi - abs(abs(phi_center - entity_phi) - math.pi)
    
    # Weighted combination
    arc_theta = d_theta * (R + r) * (1.0 - phi_weight)
    arc_phi   = (d_phi + d_phi_to_user * 0.5) * r * phi_weight * 2
    
    dist = math.sqrt(arc_theta**2 + arc_phi**2)
    max_dist = max_toroidal_distance(R, r)
    
    return (math.cos(dist / max_dist * math.pi) + 1) / 2
```
